cmake_minimum_required(VERSION 3.25)
project(FlowSketchNativeAnalyzer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
find_package(LLVM 23.1.3 EXACT REQUIRED CONFIG)
find_package(Clang REQUIRED CONFIG)
add_executable(flowsketch-analyzer main.cpp)
target_include_directories(flowsketch-analyzer SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS} ${CLANG_INCLUDE_DIRS})
target_compile_definitions(flowsketch-analyzer PRIVATE ${LLVM_DEFINITIONS} NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_link_libraries(flowsketch-analyzer PRIVATE clangTooling clangAnalysis clangFrontend clangAST clangLex clangBasic clangIndex)
if(MSVC)
  target_compile_options(flowsketch-analyzer PRIVATE /utf-8 /EHsc /bigobj /W3)
  if(NOT LLVM_ENABLE_RTTI)
    target_compile_options(flowsketch-analyzer PRIVATE /GR-)
  endif()
endif()
